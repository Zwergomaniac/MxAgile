  <script>
    // Mock data — all fictional (MOCKUP-SITE-REGISTRY, revision 2)
    let sites = [
      { name: 'Northgate Site', region: 'Northern Region', status: 'Active', dept: 'Northern' },
      { name: 'Westshore Site', region: 'Western Region',  status: 'Active', dept: 'Western'  },
      { name: 'Eastpark Site',  region: 'Eastern Region',  status: 'Inactive', dept: 'Eastern' }
    ];

    const listEl    = document.getElementById('list');
    const searchEl  = document.getElementById('search');
    const createBtn = document.getElementById('site-create-button');
    const roleEl    = document.getElementById('role');
    const modal     = document.getElementById('modal');
    const formEl    = document.getElementById('site-form');
    const nameErr   = document.getElementById('name-error');

    function isManager()  { return roleEl.value === 'manager'; }
    function isDeptMgr()  { return roleEl.value === 'deptmgr'; }

    function render() {
      const q = searchEl.value.toLowerCase();
      // ASSUM-001: Department Manager sees all until Q-002 resolved
      const filtered = sites.filter(s => s.name.toLowerCase().includes(q));

      createBtn.hidden = !isManager();

      if (filtered.length === 0) {
        listEl.innerHTML = '<p class="empty-state">No sites match your search.</p>';
        return;
      }

      listEl.innerHTML = filtered.map(s => `
        <article class="card" data-mx-req="REQ-001">
          <span class="badge">${s.status}</span>
          <h2>${s.name}</h2>
          <p>${s.region}</p>
          ${isManager() ? '<div class="actions"><button class="btn-secondary" type="button">Edit</button></div>' : ''}
          ${isDeptMgr() ? '<p style="font-size:.8em;color:#888;margin-top:6px">Scope: open question (ASSUM-001)</p>' : ''}
        </article>
      `).join('');
    }

    searchEl.addEventListener('input', render);
    roleEl.addEventListener('change', render);

    createBtn.addEventListener('click', () => {
      modal.classList.remove('hidden');
      document.getElementById('site-name').focus();
    });

    document.getElementById('cancel-btn').addEventListener('click', closeDialog);
    modal.addEventListener('click', e => { if (e.target === modal) closeDialog(); });
    document.addEventListener('keydown', e => { if (e.key === 'Escape') closeDialog(); });

    function closeDialog() {
      modal.classList.add('hidden');
      formEl.reset();
      nameErr.textContent = '';
    }

    // Save — BR-001: name required
    formEl.addEventListener('submit', e => {
      e.preventDefault();
      const name = document.getElementById('site-name').value.trim();
      if (!name) {
        nameErr.textContent = 'Site name is required (BR-001).';
        document.getElementById('site-name').focus();
        return;
      }
      const region = document.getElementById('site-region').value;
      sites.push({ name, region, status: 'Active', dept: region.replace(' Region','') });
      closeDialog();
      render();
    });

    render();
  </script>
